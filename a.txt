# 1 — instala Python 3.11 (se nao tiver)
winget install Python.Python.3.11

# 2 — reinicia o terminal depois do winget

# 3 — cria um ambiente virtual com Python 3.11 na pasta do projeto
cd C:\caminho\para\seu\projeto
py -3.11 -m venv venv_tcc

# 4 — ativa o ambiente
venv_tcc\Scripts\activate

# 5 — instala PyTorch COM CUDA (cu121 para NVIDIA com driver recente)
pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu121

# 6 — confirma que a GPU foi reconhecida
python -c "import torch; print('GPU:', torch.cuda.get_device_name(0) if torch.cuda.is_available() else 'NAO DETECTADA')"

# 7 — se imprimiu o nome da GPU, instala o resto
python setup.py
